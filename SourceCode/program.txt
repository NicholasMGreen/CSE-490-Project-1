// CSE490 board demo program - each instruction tested once
// PASS: PC 0x72 (jmp-self loop). FAIL: PC stuck at its unique FAILn address (0x74+)
// FAIL_BASE = 0x74 (must match board_top comparator and testbench!)

// T1: add -> 3 + 4 = 7
3103   // addi R1, 3
3204   // addi R2, 4
0310   // add  R3 = R1 + R2
3407   // addi R4, 7        (expected)
4431   // beq  R3, R4, +1   (equal -> skip fail jmp)
6034   // jmp FAIL1 (0x74)

// T2: sub -> 6 - 2 = 4   (checker uses bne: equal -> continue, different -> FAIL)
3506   // addi R5, 6
3602   // addi R6, 2
0651   // sub  R6 = R5 - R6
3704   // addi R7, 4        (expected)
5761   // bne  R6, R7, +1   (different -> skip to fail path)
6001   // jmp 0x1A          (equal -> continue)
602E   // jmp FAIL2 (0x76)  (reached only if sub broke)

// T3: sll -> 1 << 3 = 8
3801   // addi R8, 1
3903   // addi R9, 3
0892   // sll  R8 = R8 << R9
3A08   // addi RA, 8        (expected)
4A81   // beq  R8, RA, +1
6029   // jmp FAIL3 (0x78)

// T4: and -> 6 & 3 = 2
3B06   // addi RB, 6
3C03   // addi RC, 3
0CB3   // and  RC = RB & RC
3D02   // addi RD, 2        (expected)
4DC1   // beq  RC, RD, +1
6024   // jmp FAIL4 (0x7A)

// T5: addi with NEGATIVE immediate -> 7 + (-2) = 5  (tests sign extension)
3E07   // addi RE, 7
3EEE   // addi RE = RE + (-2)   (imm 1110 = -2)
3F05   // addi RF, 5        (expected)
4FE1   // beq  RE, RF, +1
6020   // jmp FAIL5 (0x7C)

// T6: sw + lw round trip -> store 6 at mem[4], load it back
3106   // addi R1, 6        (data)
3204   // addi R2, 4        (base address)
2120   // sw   R1 -> mem[R2+0]
1320   // lw   R3 <- mem[R2+0]
3406   // addi R4, 6        (expected)
4431   // beq  R3, R4, +1
601A   // jmp FAIL6 (0x7E)

// T7: beq taken -> equal, skips the poison
3503   // addi R5, 3
3603   // addi R6, 3
4562   // beq  R5, R6, +2   (taken -> skip 2)   <-- under test
3701   // addi R7, 1        (poison: runs only if beq failed)
6016   // jmp FAIL7 (0x80)
4701   // beq  R7, R0, +1   (R7 == 0? -> skip)
6015   // jmp FAIL7b (0x82)

// T8: bne taken -> different, skips the poison
3A03   // addi RA, 3
3B04   // addi RB, 4
5BA2   // bne  RA, RB, +2   (taken -> skip 2)   <-- under test
3C01   // addi RC, 1        (poison)
6011   // jmp FAIL8 (0x84)
5C01   // bne  R0, RC, +1   (RC != 0? -> fail path)
6001   // jmp 0x68          (RC == 0 -> continue)
600F   // jmp FAIL8b (0x86)

// T9: jmp -> skips the trap
6002   // jmp +2            (skips trap)         <-- under test
3D01   // addi RD, 1        (poison)
600D   // jmp FAIL9 (0x88)
4D01   // beq  RD, R0, +1   (RD == 0? -> skip)
600B   // jmp FAIL9 (0x88)

// success: park here (0x72) - below FAIL_BASE so the pass LED stays on
6FFF   // jmp self

// fail loops: one per check, each jmp-self - PC gets stuck, red LED on
// the stuck address tells you WHICH check failed
6FFF   // 0x74 FAIL1: add
6FFF   // 0x76 FAIL2: sub
6FFF   // 0x78 FAIL3: sll
6FFF   // 0x7A FAIL4: and
6FFF   // 0x7C FAIL5: addi
6FFF   // 0x7E FAIL6: sw/lw
6FFF   // 0x80 FAIL7: beq
6FFF   // 0x82 FAIL7b: beq (poison check)
6FFF   // 0x84 FAIL8: bne
6FFF   // 0x86 FAIL8b: bne (poison check)
6FFF   // 0x88 FAIL9: jmp
